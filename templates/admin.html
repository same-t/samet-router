Setup order

# 1) Add an upstream key from "Provider Keys":
#      provider=openai  + protocol=openai    → OpenAI / OpenAI-compatible server
#      provider=anthropic + protocol=anthropic → real Anthropic
#      provider=gemini  + protocol=gemini    → Google AI Studio (AIza...)
#      provider=custom  + base URL           → Ollama / vLLM / DeepSeek etc.
# 2) From "Models", map router name → provider + upstream model
#    (e.g.: gpt-mini → openai/gpt-4o-mini,  claude-alias → anthropic/claude-sonnet-4-5)
# 3) From "User Keys", generate a client key (sk-samet-...) and use it in the client.

Client examples

# OpenAI-compatible (python, incl. tools + stream)
from openai import OpenAI
c = OpenAI(base_url="http://localhost:8000/v1", api_key="sk-samet-...")
print(c.chat.completions.create(model="gpt-mini",
      messages=[{"role":"system","content":"Write briefly"},
                {"role":"user","content":"Hello"}]).choices[0].message.content)

# Anthropic-compatible (format used by Claude Code)
import anthropic
c = anthropic.Anthropic(base_url="http://localhost:8000", api_key="sk-samet-...")
print(c.messages.create(model="claude-alias", max_tokens=200,
      system="Write briefly",
      messages=[{"role":"user","content":"Hello"}]).content[0].text)

# Extended thinking (passed through as-is to the Anthropic provider):
print(c.messages.create(model="claude-alias", max_tokens=2000,
      thinking={"type": "enabled", "budget_tokens": 1000},
      messages=[{"role":"user","content":"Think deeply"}]).content)

# Claude Code connection (powershell):
#   $env:ANTHROPIC_BASE_URL = "http://localhost:8000"
#   $env:ANTHROPIC_AUTH_TOKEN = "sk-samet-..."  # key generated from the panel
#   $env:ANTHROPIC_MODEL = "claude-alias"
#   claude

# Gemini-compatible (raw REST)
# POST http://localhost:8000/v1beta/models/samet-flash:generateContent
# Headers: x-api-key: sk-samet-...
# {"contents":[{"role":"user","parts":[{"text":"Hello"}]}]}

Thinking support

# Anthropic thinking → Anthropic provider : as-is (signature + redacted preserved)
# Anthropic thinking → OpenAI/Gemini       : budget is carried over (reasoning_effort /
#                                            thinkingBudget), blocks cannot be carried over
# OpenAI reasoning   → Anthropic           : reasoning_content → thinking block
# Gemini thought     → Anthropic/OpenAI    : if thinking was requested, thinking block /
#                                            reasoning_content, otherwise dropped
# Unsigned thinking goes to Anthropic upstream as text (avoids 400 errors).
