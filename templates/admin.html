İstemci örnekleri

# OpenAI uyumlu (python)
from openai import OpenAI
c = OpenAI(base_url="http://localhost:8000/v1", api_key="bos-birakilabilir")
print(c.chat.completions.create(model="samet-flash",
      messages=[{"role":"user","content":"Merhaba"}]).choices[0].message.content)

# Anthropic uyumlu
import anthropic
c = anthropic.Anthropic(base_url="http://localhost:8000", api_key="x")
print(c.messages.create(model="samet-flash", max_tokens=200,
      messages=[{"role":"user","content":"Merhaba"}]).content[0].text)
